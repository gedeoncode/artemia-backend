const subCategorySchema = new mongoose.Schema(
  {
    name: {
      type: String,
      required: [true, "Subcategory name is required"],
      maxlength: [100, "Subcategory name cannot exceed 100 characters"],
      trim: true,
      lowercase: true,
    },
  },
  { _id: true, timestamps: true } // Include timestamps for subcategories
);
const categorySchema = new mongoose.Schema(
  {
    name: {
      type: String,
      required: [true, "Category name is required"],
    },
    subcategories: {
      type: [subCategorySchema],
      validate: {
        validator: function (v) {
          // Ensure subcategory names are unique (case-insensitive)
          const names = v.map((sub) => sub.name.toLowerCase());
          return new Set(names).size === names.length;
        },
        message: (props) => `Duplicate subcategory name: ${props.value}`,
      },
    },
    requiresPayment: { type: Boolean, default: false },
    isFeatured: { type: Boolean, default: false },
    isActive: { type: Boolean, default: true },
    isDeleted: { type: Boolean, default: false, select: false }, // Hidden by default in queries
    hasSubcategories: { type: Boolean, default: false },
    type: {
      type: String,
      enum: {
        values: ["physical", "digital", "service"],
        message: "Invalid category type. Must be physical, digital, or service",
      },
      default: "physical",
   
    },
    status: {
      type: String,
      enum: {
        values: ["active", "inactive", "archived"],
        message: "Status must be active, inactive, or archived",
      },
      default: "active",
   
    },
    description: {
      type: String,
    },
    image: {
      type: String,
      default: "",
    },
    slug: {
      type: String,
    },
    metadata: {
      type: mongoose.Schema.Types.Mixed,
      default: {},
    },
  },
  {
    timestamps: true,
  }
);
